    • eval_year: 2025
    • min_candidate_train_count: 15000
    • max_candidates_per_library: 300
    • rolling_train_months: 3
    • oco_include_no_touch: True
    • threshold_mode: rolling_days
    • rolling_threshold_days: 20

    • execution_quantile: 0.9
